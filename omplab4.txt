//example 1

#include <iostream>
#include <omp.h>
using namespace std;

int main() {
    

#pragma omp parallel 

#pragma omp for 
       for(int i=0;i<3;i++)
           for (int j = 0;j < 8;j++)
           {
               printf("i is %d, J is %d, in the thread number :%d \n", i, j, omp_get_thread_num());
           }

    
}

int main() {

    for(int i=0;i<4;i++)
    {
        int x = omp_get_num_threads();
        int id = omp_get_thread_num();
        printf("machine have %d threads : thread %d \n", x, id);

    }

#pragma omp critical
    {
        int x = omp_get_num_threads();
        int id = omp_get_thread_num();
        printf("machine have %d threads : thread %d \n", x, id);

    }


}


//example 6

#include <iostream>
#include <omp.h>
using namespace std;
int main() {
    int sum = 0;
    int local_sum = 0;

#pragma omp parallel
       
#pragma omp for
        for (int i = 0; i < 100; ++i) {
            local_sum += i;
        }
        
#pragma omp critical
       {

          sum += local_sum;
          cout << "local_sum : " <<sum<< endl;
       }   

}
