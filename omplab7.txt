#include<iostream>
#include<omp.h>
using namespace std;

int main() {
    omp_set_num_threads(3);
#pragma omp parallel 
    {
#pragma omp for ordered schedule(static,2)
        for (int i = 0;i < 10;i++) {
#pragma omp ordered
            printf("thread ID %d value of i is %d \n", omp_get_thread_num(), i);
        }

    }
    cout << "end loop \n";
}
